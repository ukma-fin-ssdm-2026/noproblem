1"""Бізнес-правила IAM: політики та сутності. 

2 

31. **Політики** — числа з вимог (8 символів, 5 спроб, 15 хвилин) в 

4 одному місці, а не розкидані по коду як «магічні числа». 

52. **Сутності** — дані разом із правилами, які до них стосуються 

6 (``user.is_locked(now)``, ``session.is_expired(...)``). 

7 

8Усі класи незмінні (``frozen=True``). Метод, що «змінює» об'єкт, 

9повертає *нову* копію, тож стан у сховищі змінюється лише тоді, коли 

10сервіс явно збереже цю копію. 

11"""

33@dataclass(frozen=True) 

34class PasswordPolicy: 

35 """Вимоги до пароля. 

36 

37 Attributes: 

38 min_length: Мінімальна довжина (FR-001: 8 символів). 

39 max_length: Максимальна довжина. Обмежує роботу хеш-функції на 

40 один запит, щоб довгі паролі не стали вектором DoS. 

41 """ 

42 

43 min_length: int = 8 

44 max_length: int = 128 

45 

46 def __post_init__(self) -> None: 

47 if not 1 <= self.min_length <= self.max_length: 

48 raise ValueError("Потрібно 1 <= min_length <= max_length.") 

49 

50 def check(self, password: str) -> None: 

51 """Перевіряє пароль на відповідність політиці. 

52 

53 Склад пароля (цифри, великі літери) свідомо не перевіряється: 

54 NIST SP 800-63B радить вимагати довжину, а не «складність». 

55 

56 Raises: 

57 PasswordTooShortError: Пароль коротший за ``min_length``. 

58 PasswordTooLongError: Пароль довший за ``max_length``. 

59 """ 

60 if len(password) < self.min_length: 

61 raise PasswordTooShortError(self.min_length) 

62 if len(password) > self.max_length: 

63 raise PasswordTooLongError(self.max_length)

88 

89 Два незалежні таймаути — рекомендація OWASP: сесія закінчується

95 absolute_lifetime: Максимальний строк сесії від моменту входу. 

96 touch_interval: Як часто оновлювати «останню активність». 

97 Без цього кожен запит означав би запис у сховище. 

98 """

101 absolute_lifetime: timedelta = timedelta(hours=12)

118@dataclass(frozen=True) 

119class Email: 

120 """Нормалізована email-адреса. 

121 

122 Нормалізація (обрізання пробілів і нижній регістр) потрібна, щоб 

123 ``Anna@Mail.com`` і ``anna@mail.com`` вважалися одним акаунтом 

124 (A-002: email унікальний). 

125 """ 

126 

127 value: str 

128 MAX_LENGTH: ClassVar[int] = 254 # обмеження RFC 5321 

129 

130 @classmethod 

131 def parse(cls, raw: str) -> Email: 

132 """Перевіряє та нормалізує рядок, введений користувачем. 

133 

134 Перевірка свідомо базова: існування адреси однаково 

135 підтверджує лише лист. Регулярний вираз не використовується, 

136 щоб уникнути ризику ReDoS. 

137 

138 Raises: 

139 InvalidEmailError: Рядок не схожий на email-адресу. 

140 """ 

141 value = raw.strip().lower() 

142 if not 3 <= len(value) <= cls.MAX_LENGTH: 

143 raise InvalidEmailError(value) 

144 local, separator, domain = value.partition("@") 

145 if ( 

146 not separator 

147 or not local 

148 or "@" in domain 

149 or any(char.isspace() for char in value) 

150 or "." not in domain 

151 or domain.startswith(".") 

152 or domain.endswith(".") 

153 or ".." in domain 

154 ): 

155 raise InvalidEmailError(value) 

156 return cls(value) 

157 

158 def __str__(self) -> str: 

159 return self.value

185@dataclass(frozen=True) 

186class User: 

187 """Обліковий запис користувача. 

188 

189 ``password_hash`` позначено ``repr=False``: якщо об'єкт випадково 

190 потрапить у журнал, хеш не розкриється. 

191 """ 

192 

193 id: UUID 

194 email: Email 

195 name: DisplayName 

196 password_hash: str = field(repr=False) 

197 created_at: datetime 

198 failed_login_attempts: int = 0 

199 locked_until: datetime | None = None 

200 

201 def is_locked(self, now: datetime) -> bool: 

202 """Чи заблоковано вхід у момент ``now`` (FR-005).""" 

203 return self.locked_until is not None and now < self.locked_until 

204 

205 def with_failed_login(self, now: datetime, policy: LockoutPolicy) -> User: 

206 """Повертає копію з урахованою невдалою спробою входу. 

207 

208 Після ``policy.max_failed_attempts`` спроб поспіль вхід 

209 блокується на ``policy.lock_duration``, а лічильник 

210 скидається, щоб після блокування знову було 5 спроб. 

211 """ 

212 attempts = self.failed_login_attempts + 1 

213 if attempts >= policy.max_failed_attempts: 

214 return replace( 

215 self, 

216 failed_login_attempts=0, 

217 locked_until=now + policy.lock_duration, 

218 ) 

219 return replace(self, failed_login_attempts=attempts) 

220 

221 def with_lock_cleared(self) -> User: 

222 """Повертає копію без лічильника спроб і без блокування.""" 

223 return replace(self, failed_login_attempts=0, locked_until=None) 

224 

225 def with_password_hash(self, password_hash: str) -> User: 

226 """Повертає копію з новим хешем пароля.""" 

227 return replace(self, password_hash=password_hash)

230@dataclass(frozen=True) 

231class Session: 

232 """Сесія автентифікованого користувача. 

233 

234 Сам токен ніде не зберігається, лише його SHA-256 відбиток 

235 (``token_hash``). Якщо сховище скомпрометують, діючих токенів з 

236 нього отримати не можна. 

237 """ 

238 

239 id: UUID 

240 user_id: UUID 

241 token_hash: str = field(repr=False) 

242 created_at: datetime 

243 last_seen_at: datetime 

244 expires_at: datetime 

245 

246 def is_expired(self, now: datetime, policy: SessionPolicy) -> bool: 

247 """Чи закінчилася сесія (абсолютний строк або бездіяльність).""" 

248 return ( 

249 now >= self.expires_at 

250 or now - self.last_seen_at >= policy.idle_timeout 

251 ) 

252 

253 def needs_touch(self, now: datetime, policy: SessionPolicy) -> bool: 

254 """Чи час оновити позначку останньої активності.""" 

255 return now - self.last_seen_at >= policy.touch_interval 

256 

257 def touched(self, now: datetime) -> Session: 

258 """Повертає копію з оновленою позначкою активності.""" 

259 return replace(self, last_seen_at=now)
