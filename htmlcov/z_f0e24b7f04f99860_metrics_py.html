39class MetricsMiddleware(BaseHTTPMiddleware): 

40 async def dispatch(self, request: Request, call_next): 

41 if request.url.path == "/metrics": 

42 return await call_next(request) 

43 

44 start = time.perf_counter() 

45 status_code = 500 

46 size = 0 

47 try: 

48 response = await call_next(request) 

49 status_code = response.status_code 

50 size = int(response.headers.get("content-length", 0)) 

51 return response 

52 finally: 

53 elapsed = time.perf_counter() - start 

54 route = getattr(request.scope.get("route"), "path", "unmatched") 

55 HTTP_REQUESTS.labels(request.method, route, str(status_code)).inc() 

56 HTTP_DURATION.labels(request.method, route).observe(elapsed) 

57 HTTP_RESPONSE_SIZE.labels(request.method, route).observe(size)
